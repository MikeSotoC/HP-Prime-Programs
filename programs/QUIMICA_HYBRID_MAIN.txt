QH_REDOX()
BEGIN
 LOCAL opcionRedox;
 LOCAL sustanciaRedox;
 opcionRedox:=1;
 sustanciaRedox:="";
 REPEAT
  opcionRedox:=QH_MENU("EDITOR DE ECUACION","BALANCEO Y EXPLICACION",{"AGREGAR REACTIVO","PASAR A PRODUCTOS","AGREGAR PRODUCTO","VER ECUACION","BALANCEAR Y EXPLICAR","BORRAR TODO"});
  IF opcionRedox==1 THEN
   sustanciaRedox:=QH_ESPECIE();
   IF SIZE(sustanciaRedox)>0 THEN QREACT:=QREACT+IFTE(SIZE(QREACT)>0,"|","")+sustanciaRedox; QRSHOW:=QRSHOW+IFTE(SIZE(QRSHOW)>0," + ","")+QLASTSHOW; END;
  END;
  IF opcionRedox==2 THEN
   IF SIZE(QREACT)==0 THEN QH_AVISO("Primero agregue al menos un reactivo."); ELSE QH_AVISO("Ahora agregue los productos."); END;
  END;
  IF opcionRedox==3 THEN
   IF SIZE(QREACT)==0 THEN QH_AVISO("Primero agregue los reactivos."); ELSE
    sustanciaRedox:=QH_ESPECIE();
    IF SIZE(sustanciaRedox)>0 THEN QPROD:=QPROD+IFTE(SIZE(QPROD)>0,"|","")+sustanciaRedox; QPSHOW:=QPSHOW+IFTE(SIZE(QPSHOW)>0," + ","")+QLASTSHOW; END;
   END;
  END;
  IF opcionRedox==4 THEN QH_VER_ECUACION(); END;
  IF opcionRedox==5 THEN
   IF SIZE(QREACT)==0 OR SIZE(QPROD)==0 THEN QH_AVISO("Complete ambos lados de la ecuacion."); ELSE
    QRXN:=QREACT+">"+QPROD; QOP:=2001; QERR:=0; QTEXT:=""; PYTHON("import CHEM_ENGINE");
    IF QERR THEN QH_AVISO(QTEXT); ELSE QH_RESULT("BALANCEO REDOX",QTEXT); END;
   END;
  END;
  IF opcionRedox==6 THEN QREACT:=""; QPROD:=""; QRSHOW:=""; QPSHOW:=""; QH_AVISO("Ecuacion borrada."); END;
 UNTIL opcionRedox==0;
END;


QH_ESPECIE()
BEGIN
 LOCAL textoFormula;
 LOCAL textoVisible;
 LOCAL opcionEspecie;
 LOCAL numeroAtomico;
 LOCAL subindice;
 LOCAL indiceRadical;
 LOCAL cargaIon;
 LOCAL listaSimbolos;
 LOCAL parteGrupo;
 textoFormula:="";
 textoVisible:="";
 opcionEspecie:=1;
 subindice:=1;
 cargaIon:=0;
 parteGrupo:="";
 listaSimbolos:=TP_SIMBOLOS();
 REPEAT
  opcionEspecie:=QH_MENU("CONSTRUIR SUSTANCIA","ELEMENTOS, GRUPOS Y CARGA",{"AGREGAR ELEMENTO","AGREGAR RADICAL O GRUPO","FINALIZAR NEUTRA","FINALIZAR COMO ION","REINICIAR SUSTANCIA"});
  IF opcionEspecie==1 THEN
   numeroAtomico:=TP_SELECCIONAR("ELEMENTO DE LA SUSTANCIA");
   IF numeroAtomico>0 THEN
    subindice:=1; IF INPUT(subindice,"SUBINDICE","Cantidad de "+listaSimbolos(numeroAtomico),"1 no se escribe") THEN
     IF subindice>=1 AND IP(subindice)==subindice THEN textoFormula:=textoFormula+listaSimbolos(numeroAtomico)+IFTE(subindice==1,"",STRING(subindice)); textoVisible:=textoVisible+listaSimbolos(numeroAtomico)+IFTE(subindice==1,"",STRING(subindice)); ELSE QH_AVISO("El subindice debe ser entero positivo."); END;
    END;
   END;
  END;
  IF opcionEspecie==2 THEN
   indiceRadical:=QH_RADICALES();
   IF indiceRadical>0 THEN
    QRID:=indiceRadical; QOP:=1003; QERR:=0; QFORM:=""; PYTHON("import CHEM_ENGINE");
    IF QERR THEN QH_AVISO(QTEXT); ELSE
     subindice:=1; IF INPUT(subindice,"SUBINDICE DEL GRUPO","Cantidad de "+QFORM,"1 no se escribe") THEN
      IF subindice>=1 AND IP(subindice)==subindice THEN parteGrupo:=IFTE(subindice==1,QFORM,"("+QFORM+")"+STRING(subindice)); textoFormula:=textoFormula+parteGrupo; textoVisible:=textoVisible+parteGrupo; ELSE QH_AVISO("El subindice debe ser entero positivo."); END;
     END;
    END;
   END;
  END;
  IF opcionEspecie==3 THEN
   IF SIZE(textoFormula)>0 THEN QLASTSHOW:=textoVisible; RETURN textoFormula; ELSE QH_AVISO("Agregue al menos un elemento."); END;
  END;
  IF opcionEspecie==4 THEN
   IF SIZE(textoFormula)>0 THEN
    cargaIon:=-1;
    IF INPUT(cargaIon,"CARGA TOTAL DEL ION","Carga, no oxidacion","Ejemplo: -2 o +1") THEN
     IF cargaIon<>0 AND IP(cargaIon)==cargaIon THEN QLASTSHOW:="["+textoVisible+"]^"+IFTE(cargaIon>0,"+","")+STRING(cargaIon); RETURN textoFormula+"{"+STRING(cargaIon)+"}"; ELSE QH_AVISO("La carga debe ser un entero distinto de cero."); END;
    END;
   ELSE QH_AVISO("Agregue al menos un elemento.");
   END;
  END;
  IF opcionEspecie==5 THEN textoFormula:=""; textoVisible:=""; END;
 UNTIL opcionEspecie==0;
 RETURN "";
END;

QH_NOMENCLATURA()
BEGIN
 LOCAL modo:=1;
 LOCAL z1;
 LOCAL z2;
 LOCAL v1:=2;
 LOCAL v2:=-2;
 REPEAT
  modo:=QH_MENU("NOMENCLATURA","FORMULAR, INTERPRETAR Y NOMBRAR",{"COMPUESTO BINARIO","HIDROXIDO U OXOSAL","OXOACIDO","SUSTANCIA SIMPLE","CATALOGO DE RADICALES","CATALOGO DE ANIONES -URO"});
  IF modo==1 THEN
   z1:=TP_SELECCIONAR("PRIMER ELEMENTO");
   IF z1>0 THEN
    z2:=TP_SELECCIONAR("SEGUNDO ELEMENTO");
    IF z2>0 THEN
     IF INPUT({v1,v2},"NUMEROS DE OXIDACION",{"Primer elemento:","Segundo elemento:"},{"Use + para el positivo","Use - para el negativo"},{2,-2}) THEN
      QZ1:=z1; QZ2:=z2; QV1:=v1; QV2:=v2;
      QH_RUN_NOM();
     END;
    END;
   END;
  END;
  IF modo==2 THEN QH_GRUPO(1); END;
  IF modo==3 THEN QH_GRUPO(2); END;
  IF modo==4 THEN QH_SIMPLE(); END;
  IF modo==5 THEN QH_CATALOGO_RADICALES(); END;
  IF modo==6 THEN QH_CATALOGO_URO(); END;
 UNTIL modo==0;
END;

QH_SIMPLE()
BEGIN
 LOCAL z:=TP_SELECCIONAR("ELEMENTO PURO");
 LOCAL s;
 LOCAL n;
 LOCAL ss:=TP_SIMBOLOS();
 LOCAL nn:=TP_NOMBRES();
 IF z==0 THEN RETURN; END;
 s:=ss(z); n:=nn(z);
 QH_DIBUJAR_FORMULA(s,1,0,"",1,0);
 QH_RESULT("SUSTANCIA SIMPLE","1. INTERPRETACION"+CHAR(10)+"Formula: "+s+CHAR(10)+"Funcion: SUSTANCIA SIMPLE"+CHAR(10)+CHAR(10)+"2. OXIDACION"+CHAR(10)+"Todo elemento libre tiene numero de oxidacion 0."+CHAR(10)+s+" = 0"+CHAR(10)+CHAR(10)+"3. NOMBRE"+CHAR(10)+n);
END;

QH_SUPER(numero)
BEGIN
 LOCAL texto:=STRING(numero);
 LOCAL salida:="";
 LOCAL indice;
 LOCAL digito;
 LOCAL superiores:={"⁰","¹","²","³","⁴","⁵","⁶","⁷","⁸","⁹"};
 FOR indice FROM 1 TO SIZE(texto) DO
  digito:=INSTRING("0123456789",MID(texto,indice,1))-1;
  salida:=salida+superiores(digito+1);
 END;
 RETURN salida;
END;

QH_CONFIG_ELECTRONICA(numeroAtomico,encabezado)
BEGIN
 LOCAL listaN:={1,2,2,3,3,4,3,4,5,4,5,6,4,5,6,7,5,6,7};
 LOCAL listaL:={"s","s","p","s","p","s","d","p","s","d","p","s","f","d","p","s","f","d","p"};
 LOCAL listaCap:={2,2,6,2,6,2,10,6,2,10,6,2,14,10,6,2,14,10,6};
 LOCAL restante:=numeroAtomico;
 LOCAL configuracion:="";
 LOCAL niveles:={};
 LOCAL letras:={};
 LOCAL cantidades:={};
 LOCAL indice;
 LOCAL electrones;
 LOCAL periodo:=0;
 LOCAL grupo:=0;
 LOCAL valencia:=0;
 LOCAL ultimoNivel:=0;
 LOCAL ultimaLetra:="";
 LOCAL ultimoElectron:=0;
 LOCAL simbolos:=TP_SIMBOLOS();
 FOR indice FROM 1 TO SIZE(listaN) DO
  IF restante>0 THEN
   electrones:=MIN(restante,listaCap(indice));
   IF SIZE(configuracion)>0 THEN configuracion:=configuracion+" "; END;
   configuracion:=configuracion+STRING(listaN(indice))+listaL(indice)+QH_SUPER(electrones);
   niveles:=CONCAT(niveles,{listaN(indice)});
   letras:=CONCAT(letras,{listaL(indice)});
   cantidades:=CONCAT(cantidades,{electrones});
   restante:=restante-electrones;
  END;
 END;
 ultimoNivel:=niveles(SIZE(niveles));
 ultimaLetra:=letras(SIZE(letras));
 ultimoElectron:=cantidades(SIZE(cantidades));
 FOR indice FROM 1 TO SIZE(niveles) DO
  IF niveles(indice)>periodo THEN periodo:=niveles(indice); END;
 END;
 FOR indice FROM 1 TO SIZE(niveles) DO
  IF niveles(indice)==periodo THEN valencia:=valencia+cantidades(indice); END;
 END;
 IF ultimaLetra=="s" THEN grupo:=ultimoElectron; END;
 IF ultimaLetra=="p" THEN grupo:=ultimoElectron+12; END;
 IF ultimaLetra=="d" THEN grupo:=ultimoElectron+2; END;
 IF ultimaLetra=="f" THEN grupo:=3; END;
 IF numeroAtomico==2 THEN grupo:=18; END;
 QH_RESULT(encabezado,
  "1. ELEMENTO"+CHAR(10)+simbolos(numeroAtomico)+"   Z = "+STRING(numeroAtomico)+CHAR(10)+CHAR(10)+
  "2. REGLA DE LLENADO"+CHAR(10)+"Aufbau + Pauli + Hund"+CHAR(10)+CHAR(10)+
  "3. CONFIGURACION"+CHAR(10)+configuracion+CHAR(10)+CHAR(10)+
  "4. ANALISIS"+CHAR(10)+"Grupo = "+STRING(grupo)+CHAR(10)+"Periodo = "+STRING(periodo)+CHAR(10)+
  "Electrones de valencia = "+STRING(valencia)+CHAR(10)+"Familia = "+TP_FAMILIA(numeroAtomico,grupo)+CHAR(10)+
  "Bloque = "+TP_BLOQUE(numeroAtomico,grupo));
END;

QH_LINEAS(texto)
BEGIN
 LOCAL salida:={};
 LOCAL resto:=texto;
 LOCAL pos;
 LOCAL linea;
 REPEAT
  pos:=INSTRING(resto,CHAR(10));
  IF pos==0 THEN
   linea:=resto; resto:="";
  ELSE
   IF pos==1 THEN linea:=" "; ELSE linea:=LEFT(resto,pos-1); END;
   resto:=MID(resto,pos+1,SIZE(resto)-pos);
  END;
  salida:=CONCAT(salida,{linea});
 UNTIL SIZE(resto)==0;
 RETURN salida;
END;

TP_CLASE(z)
BEGIN
 IF z==1 THEN RETURN "No metal"; END;
 IF z==5 OR z==14 OR z==32 OR z==33 OR z==51 OR z==52 OR z==84 THEN RETURN "Metaloide"; END;
 IF z==6 OR z==7 OR z==8 OR z==9 OR z==15 OR z==16 OR z==17 OR z==34 OR z==35 OR z==53 OR z==85 OR z==117 THEN RETURN "No metal"; END;
 IF z==2 OR z==10 OR z==18 OR z==36 OR z==54 OR z==86 OR z==118 THEN RETURN "Gas noble"; END;
 RETURN "Metal";
END;

TP_FAMILIA(z,g)
BEGIN
 IF z>=57 AND z<=71 THEN RETURN "Lantanido"; END;
 IF z>=89 AND z<=103 THEN RETURN "Actinido"; END;
 IF g==1 THEN RETURN IFTE(z==1,"No metal","Alcalino"); END;
 IF g==2 THEN RETURN "Alcalinoterreo"; END;
 IF g>=3 AND g<=12 THEN RETURN "Metal de transicion"; END;
 IF g==13 THEN RETURN "Familia del boro"; END;
 IF g==14 THEN RETURN "Familia del carbono"; END;
 IF g==15 THEN RETURN "Nitrogenoideos"; END;
 IF g==16 THEN RETURN "Calcogenos"; END;
 IF g==17 THEN RETURN "Halogenos"; END;
 IF g==18 THEN RETURN "Gases nobles"; END;
 RETURN "Bloque f";
END;

TP_BLOQUE(z,g)
BEGIN
 IF z>=57 AND z<=71 OR z>=89 AND z<=103 THEN RETURN "f"; END;
 IF g<=2 THEN RETURN "s"; END;
 IF g>=13 THEN RETURN "p"; END;
 RETURN "d";
END;

TP_SELECCIONAR(titulo)
BEGIN
 LOCAL simbolos:=TP_SIMBOLOS();
 LOCAL nombres:=TP_NOMBRES();
 LOCAL grupos:=TP_GRUPOS();
 LOCAL periodos:=TP_PERIODOS();
 LOCAL sel:=1;
 LOCAL nuevo:=1;
 LOCAL k:=-1;
 LOCAL x:=-1;
 LOCAL y:=-1;
 LOCAL z;
 LOCAL pos;
 LOCAL cx;
 LOCAL cy;
 TP_DIBUJAR_TABLA(sel,simbolos,nombres,grupos,periodos);
 RECT_P(0,0,319,22,#212728,#212728);
 TEXTOUT_P(titulo+": "+simbolos(sel),5,4,1,#FFFFFF);
 TP_LIMPIAR_BUFFER();
 REPEAT
  k:=GETKEY; x:=MOUSE(0); y:=MOUSE(1); nuevo:=sel;
  IF k==7 THEN nuevo:=TP_MOVER(sel,7,grupos,periodos); END;
  IF k==8 THEN nuevo:=TP_MOVER(sel,8,grupos,periodos); END;
  IF k==2 THEN nuevo:=TP_MOVER(sel,2,grupos,periodos); END;
  IF k==12 THEN nuevo:=TP_MOVER(sel,12,grupos,periodos); END;
  IF k==30 THEN RETURN sel; END;
  IF k==4 OR k==19 THEN RETURN 0; END;
  IF x>=0 AND y>=216 THEN
   IF x<=104 THEN RETURN 0; END;
   IF x>=211 THEN RETURN sel; END;
  END;
  IF x>=0 AND y>=25 AND y<=205 THEN
   FOR z FROM 1 TO 118 DO
    pos:=TP_POS(z,grupos,periodos);
    cx:=2+(pos(1)-1)*17; cy:=25+(pos(2)-1)*20;
    IF x>=cx AND x<=cx+15 AND y>=cy AND y<=cy+17 THEN nuevo:=z; END;
   END;
  END;
  IF nuevo<>sel THEN
   TP_DIBUJAR_CELDA(sel,0,simbolos,grupos,periodos);
   sel:=nuevo;
   TP_DIBUJAR_CELDA(sel,1,simbolos,grupos,periodos);
   RECT_P(0,0,319,22,#212728,#212728);
   TEXTOUT_P(titulo+": "+simbolos(sel)+" "+nombres(sel),5,4,1,#FFFFFF);
  END;
  WAIT(0.05);
 UNTIL k==-999;
END;

TP_MOVER(actual,direccion,listaGrupos,listaPeriodos)
BEGIN
 LOCAL posicionActual:=TP_POS(actual,listaGrupos,listaPeriodos);
 LOCAL posicionCandidata;
 LOCAL candidato;
 LOCAL mejor:=actual;
 LOCAL puntaje:=999;
 LOCAL nuevoPuntaje;
 FOR candidato FROM 1 TO 118 DO
  posicionCandidata:=TP_POS(candidato,listaGrupos,listaPeriodos);
  IF direccion==7 AND posicionCandidata(2)==posicionActual(2) AND posicionCandidata(1)<posicionActual(1) THEN nuevoPuntaje:=posicionActual(1)-posicionCandidata(1); IF nuevoPuntaje<puntaje THEN puntaje:=nuevoPuntaje; mejor:=candidato; END; END;
  IF direccion==8 AND posicionCandidata(2)==posicionActual(2) AND posicionCandidata(1)>posicionActual(1) THEN nuevoPuntaje:=posicionCandidata(1)-posicionActual(1); IF nuevoPuntaje<puntaje THEN puntaje:=nuevoPuntaje; mejor:=candidato; END; END;
  IF direccion==2 AND posicionCandidata(2)<posicionActual(2) THEN nuevoPuntaje:=ABS(posicionCandidata(1)-posicionActual(1))*3+posicionActual(2)-posicionCandidata(2); IF nuevoPuntaje<puntaje THEN puntaje:=nuevoPuntaje; mejor:=candidato; END; END;
  IF direccion==12 AND posicionCandidata(2)>posicionActual(2) THEN nuevoPuntaje:=ABS(posicionCandidata(1)-posicionActual(1))*3+posicionCandidata(2)-posicionActual(2); IF nuevoPuntaje<puntaje THEN puntaje:=nuevoPuntaje; mejor:=candidato; END; END;
 END;
 RETURN mejor;
END;

// Corta un texto en lineas de maximo n caracteres (por espacios).
QH_ENVOLVER(t,n)
BEGIN
 LOCAL salida:={};
 LOCAL resto:=t;
 LOCAL k;
 WHILE SIZE(resto)>n DO
  k:=n;
  WHILE k>1 AND MID(resto,k,1)<>" " DO k:=k-1; END;
  IF k==1 THEN k:=n; END;
  salida:=CONCAT(salida,{LEFT(resto,k)});
  resto:=MID(resto,k+1,SIZE(resto)-k);
 END;
 RETURN CONCAT(salida,{resto});
END;

// Tarjeta de aviso NO bloqueante: aparece abajo, se cierra sola
// (o con cualquier tecla/toque) y la pantalla de fondo se mantiene.

BEGIN

 LOCAL n:=MIN(3,SIZE(lineas));
 LOCAL h:=n*16+16;
 LOCAL y0:=208-h;
 LOCAL color:=#217B9C;

 LOCAL fin:=0;
 IF tipo==1 THEN color:=#2E7D5B; simbolo:="OK"; END;
 IF tipo==2 THEN color:=#C98A1B; simbolo:="!"; END;
 IF tipo==3 THEN color:=#B3261E; simbolo:="X"; END;
 RECT_P(8,y0-2,311,210,color,#FFFFFF);
 RECT_P(8,y0-2,50,210,color,color);
 TEXTOUT_P(simbolo,IFTE(SIZE(simbolo)==2,17,23),y0+IP(h/2)-8,2,#FFFFFF);
 FOR indiceLinea FROM 1 TO n DO
  TEXTOUT_P(lineas(indiceLinea),58,y0+6+(indiceLinea-1)*16,1,#142837);
 END;
 TP_LIMPIAR_BUFFER();
 REPEAT
  WAIT(0.05);
  ciclos:=ciclos+1;
  IF GETKEY<>-1 OR MOUSE(0)>=0 THEN fin:=1; END;
 UNTIL fin==1 OR ciclos>26+12*n;
 WAIT(0.2);
END;

// activos : lista de 0/1 (misma longitud que opciones) para mostrar la
//           insignia "ACTIVO"; use {} si no hace falta.
// inicial : opcion que aparece seleccionada al abrir.
// aviso   : si no esta vacio, muestra una tarjeta de exito SIN salir del menu.
QH_MENU2(titulo,subtitulo,opciones,activos,inicial,aviso)
BEGIN
 LOCAL total:=SIZE(opciones);
 LOCAL porPagina:=4;
 LOCAL primera:=1;
 LOCAL primeraAnterior:=1;
 LOCAL seleccionado:=MAX(1,MIN(total,inicial));
 LOCAL anterior:=1;
 LOCAL tecla:=-1;
 LOCAL toqueX:=-1;
 LOCAL toqueY:=-1;
 LOCAL toqueActivo:=0;
 LOCAL indice;
 LOCAL inicioY:=48;
 LOCAL alto:=35;
 LOCAL espacio:=6;
 LOCAL posY;
 WHILE SIZE(activos)<total DO
  activos:=CONCAT(activos,{0});
 END;
 IF total==1 THEN inicioY:=82; alto:=56; END;
 IF total==2 THEN inicioY:=67; alto:=56; espacio:=14; END;
 IF total==3 THEN inicioY:=52; alto:=43; espacio:=9; END;

 QH_MENU_FONDO(titulo,subtitulo);
 QH_MENU_LISTA(opciones,activos,seleccionado,primera,inicioY,alto,espacio);
 QH_MENU_CONTADOR(seleccionado,total);
 QH_MENU_PIE("VOLVER","FLECHAS","ABRIR");
 IF SIZE(aviso)>0 THEN
  QH_TOAST(aviso,1);
  QH_MENU_LISTA(opciones,activos,seleccionado,primera,inicioY,alto,espacio);
 END;
 TP_LIMPIAR_BUFFER();
 REPEAT
  tecla:=GETKEY;
  toqueX:=MOUSE(0);
  toqueY:=MOUSE(1);
  IF toqueX<0 THEN toqueActivo:=0; END;
  anterior:=seleccionado;
  primeraAnterior:=primera;
  IF tecla==2 OR tecla==7 THEN seleccionado:=MAX(1,seleccionado-1); END;
  IF tecla==12 OR tecla==8 THEN seleccionado:=MIN(total,seleccionado+1); END;
  IF seleccionado<primera THEN primera:=seleccionado; END;
  IF seleccionado>primera+porPagina-1 THEN primera:=seleccionado-porPagina+1; END;
  IF anterior<>seleccionado THEN
   IF primera<>primeraAnterior THEN
    QH_MENU_LISTA(opciones,activos,seleccionado,primera,inicioY,alto,espacio);
   ELSE
    QH_MENU_TARJETA(anterior,opciones(anterior),0,anterior-primera+1,inicioY,alto,espacio,activos(anterior));
    QH_MENU_TARJETA(seleccionado,opciones(seleccionado),1,seleccionado-primera+1,inicioY,alto,espacio,activos(seleccionado));
   END;
   QH_MENU_CONTADOR(seleccionado,total);
  END;
  IF tecla==30 THEN RETURN seleccionado; END;
  IF tecla==4 OR tecla==19 THEN RETURN 0; END;
  IF toqueX>=0 AND toqueActivo==0 THEN
   toqueActivo:=1;
   IF toqueY>=216 AND toqueX<=104 THEN RETURN 0; END;
   IF toqueY>=216 AND toqueX>=211 THEN RETURN seleccionado; END;
   FOR indice FROM primera TO MIN(total,primera+porPagina-1) DO
    posY:=inicioY+(indice-primera)*(alto+espacio);
    IF toqueY>=posY AND toqueY<=posY+alto THEN
     IF seleccionado==indice THEN RETURN indice; END;
     anterior:=seleccionado;
     seleccionado:=indice;
     QH_MENU_TARJETA(anterior,opciones(anterior),0,anterior-primera+1,inicioY,alto,espacio,activos(anterior));
     QH_MENU_TARJETA(seleccionado,opciones(seleccionado),1,seleccionado-primera+1,inicioY,alto,espacio,activos(seleccionado));
     QH_MENU_CONTADOR(seleccionado,total);
    END;
   END;
  END;
  WAIT(0.04);
 UNTIL tecla==-999;
END;

// Ajusta lineas largas (ya no se cortan en el borde de la pantalla).
QH_AJUSTAR(lineas)
BEGIN
 LOCAL salida:={};
 LOCAL indiceLinea;
 FOR indiceLinea FROM 1 TO SIZE(lineas) DO
  IF SIZE(lineas(indiceLinea))>36 THEN
   salida:=CONCAT(salida,QH_ENVOLVER(lineas(indiceLinea),36));
  ELSE
   salida:=CONCAT(salida,{lineas(indiceLinea)});
  END;
 END;
 RETURN salida;
END;

QH_RESULT(titulo,texto)
BEGIN
 LOCAL lineas:=QH_AJUSTAR(QH_LINEAS(texto));
 LOCAL primero:=1;
 LOCAL maximo:=MAX(1,SIZE(lineas)-7);
 LOCAL indiceLinea;
 LOCAL y;
 LOCAL tecla:=-1;
 LOCAL mx;
 LOCAL my;
 LOCAL tocando:=0;
 LOCAL ultimoY:=0;
 LOCAL nuevo;
 LOCAL redibujar:=1;
 TP_LIMPIAR_BUFFER();
 REPEAT
  IF redibujar THEN
   RECT_P(#E2E7E8);
   RECT_P(0,0,319,38,#212728,#212728);
   RECT_P(0,38,319,41,#217B9C,#217B9C);
   TEXTOUT_P(titulo,10,8,2,#FFFFFF);
   FOR indiceLinea FROM 0 TO 7 DO
    IF primero+indiceLinea<=SIZE(lineas) THEN
     y:=50+indiceLinea*20;
     IF QH_ES_SECCION(lineas(primero+indiceLinea)) THEN
      RECT_P(4,y-3,300,y+14,#CFE3EC,#CFE3EC);
      TEXTOUT_P(lineas(primero+indiceLinea),10,y,1,#175E78,290,#CFE3EC);
     ELSE
      TEXTOUT_P(lineas(primero+indiceLinea),14,y,1,#142837,286,#E2E7E8);
     END;
    END;
   END;
   RECT_P(306,45,318,211,#D3DADB,#D3DADB);
   y:=45+IP((primero-1)*146/MAX(1,maximo-1));
   RECT_P(308,y,316,MIN(211,y+20),#217B9C,#217B9C);
   RECT_P(0,216,79,239,#252C2E,#252C2E);
   RECT_P(81,216,160,239,#394345,#394345);
   RECT_P(162,216,205,239,#D3DADB,#D3DADB);
   RECT_P(207,216,319,239,#217B9C,#217B9C);
   TEXTOUT_P("SUBIR",17,222,1,#FFFFFF);
   TEXTOUT_P("BAJAR",96,222,1,#FFFFFF);
   TEXTOUT_P(STRING(primero)+"/"+STRING(maximo),166,222,0,#586668);
   TEXTOUT_P("< VOLVER",232,222,1,#FFFFFF);
   redibujar:=0;
  END;
  tecla:=GETKEY; mx:=MOUSE(0); my:=MOUSE(1);
  nuevo:=primero;
  IF tecla==2 OR tecla==7 THEN nuevo:=MAX(1,primero-1); END;
  IF tecla==12 OR tecla==8 THEN nuevo:=MIN(maximo,primero+1); END;
  IF mx>=0 AND my>=216 THEN
   IF mx<=79 THEN nuevo:=MAX(1,primero-1); END;
   IF mx>=81 AND mx<=160 THEN nuevo:=MIN(maximo,primero+1); END;
   IF mx>=207 THEN tecla:=4; END;
  END;
  IF mx>=0 AND my<214 THEN
   IF tocando==0 THEN tocando:=1; ultimoY:=my; ELSE
    IF ABS(my-ultimoY)>=12 THEN nuevo:=MAX(1,MIN(maximo,primero+IFTE(my<ultimoY,1,-1))); ultimoY:=my; END;
   END;
  ELSE tocando:=0; END;
  IF nuevo<>primero THEN primero:=nuevo; redibujar:=1; END;
  WAIT(0.03);
 UNTIL tecla==4 OR tecla==19 OR tecla==30;
END;
